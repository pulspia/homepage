# Pulspia 관리자 페이지 개발 계획

## 1. 프로젝트 개요

### 1.1 목적
- 홈페이지 운영 현황 실시간 모니터링
- 서버 및 시스템 상태 관리
- 보안 및 접속 제어
- 데이터 기반 의사결정 지원

### 1.2 접근 방식
- URL: `https://pulspia.com/admin`
- GUI 버튼 없이 URL 직접 접근
- 기존 Supabase 인증 구조 활용

## 2. 기술 스택

### 2.1 프론트엔드
- **React 19**: 기존 홈페이지와 동일한 버전
- **React Router v7**: `/admin` 라우트 추가
- **Recharts**: 통계 그래프 시각화
- **Tailwind CSS**: 기존 디자인 시스템 통일
- **Lucide React**: 아이콘 라이브러리

### 2.2 백엔드 & 인프라
- **Supabase**:
  - 관리자 인증 (기존 구조 재사용)
  - 데이터베이스 (통계, 로그, 설정 저장)
  - Realtime 구독 (실시간 접속자 모니터링)
- **Node.js 모니터링 서버**:
  - 서버 리소스 모니터링 (CPU, RAM, Disk)
  - 시스템 로그 수집
  - 포트: 8002 (별도 백엔드)
- **EmailJS**: 알람 발송 (기존 Contact 구조 재사용)

### 2.3 모니터링 & 분석
- **Google Analytics 4 (GA4)**: 방문자 통계 데이터 수집
- **Custom Tracking**: 실시간 접속자, 페이지별 활동

## 4. 데이터베이스 스키마 설계

### 4.1 Supabase 테이블 구조

#### `admin_settings` (관리자 설정)
```sql
CREATE TABLE admin_settings (
  id UUID PRIMARY KEY DEFAULT uuid_generate_v4(),
  site_name TEXT DEFAULT 'Pulspia',
  admin_email TEXT NOT NULL,
  seo_keywords TEXT[],
  alert_email TEXT NOT NULL,
  cpu_threshold INTEGER DEFAULT 80,  -- CPU 임계치 (%)
  ram_threshold INTEGER DEFAULT 80,  -- RAM 임계치 (%)
  disk_threshold INTEGER DEFAULT 85, -- Disk 임계치 (%)
  traffic_threshold BIGINT DEFAULT 1073741824, -- 트래픽 임계치 (1GB)
  created_at TIMESTAMP WITH TIME ZONE DEFAULT NOW(),
  updated_at TIMESTAMP WITH TIME ZONE DEFAULT NOW()
);
```

#### `visitor_analytics` (방문자 통계)
```sql
CREATE TABLE visitor_analytics (
  id UUID PRIMARY KEY DEFAULT uuid_generate_v4(),
  session_id TEXT UNIQUE NOT NULL,
  user_agent TEXT,
  ip_address INET,
  referrer TEXT,
  landing_page TEXT,
  device_type TEXT CHECK (device_type IN ('desktop', 'mobile', 'tablet')),
  browser TEXT,
  os TEXT,
  country TEXT,
  visited_at TIMESTAMP WITH TIME ZONE DEFAULT NOW(),
  left_at TIMESTAMP WITH TIME ZONE
);
```

#### `page_views` (페이지 뷰)
```sql
CREATE TABLE page_views (
  id UUID PRIMARY KEY DEFAULT uuid_generate_v4(),
  session_id TEXT NOT NULL REFERENCES visitor_analytics(session_id),
  page_path TEXT NOT NULL,
  viewed_at TIMESTAMP WITH TIME ZONE DEFAULT NOW(),
  duration_seconds INTEGER
);
```

#### `system_logs` (시스템 로그)
```sql
CREATE TABLE system_logs (
  id UUID PRIMARY KEY DEFAULT uuid_generate_v4(),
  log_type TEXT CHECK (log_type IN ('error', 'warning', 'info')),
  error_code INTEGER, -- 404, 500 등
  message TEXT NOT NULL,
  stack_trace TEXT,
  user_agent TEXT,
  ip_address INET,
  page_url TEXT,
  created_at TIMESTAMP WITH TIME ZONE DEFAULT NOW()
);
```

#### `server_metrics` (서버 메트릭)
```sql
CREATE TABLE server_metrics (
  id UUID PRIMARY KEY DEFAULT uuid_generate_v4(),
  cpu_usage DECIMAL(5,2),  -- 0.00 ~ 100.00
  ram_usage DECIMAL(5,2),
  ram_total BIGINT,
  ram_used BIGINT,
  disk_usage DECIMAL(5,2),
  disk_total BIGINT,
  disk_used BIGINT,
  network_rx BIGINT, -- 수신 바이트
  network_tx BIGINT, -- 송신 바이트
  recorded_at TIMESTAMP WITH TIME ZONE DEFAULT NOW()
);
```

#### `ip_blacklist` (IP 차단 목록)
```sql
CREATE TABLE ip_blacklist (
  id UUID PRIMARY KEY DEFAULT uuid_generate_v4(),
  ip_address INET UNIQUE NOT NULL,
  reason TEXT,
  blocked_by UUID REFERENCES auth.users(id),
  blocked_at TIMESTAMP WITH TIME ZONE DEFAULT NOW(),
  expires_at TIMESTAMP WITH TIME ZONE
);
```

#### `admin_activity_logs` (관리자 활동 로그)
```sql
CREATE TABLE admin_activity_logs (
  id UUID PRIMARY KEY DEFAULT uuid_generate_v4(),
  admin_id UUID REFERENCES auth.users(id),
  action_type TEXT NOT NULL, -- 'create', 'update', 'delete', 'login', 'block_ip' 등
  target_table TEXT,
  target_id TEXT,
  changes JSONB, -- 변경 전후 데이터
  ip_address INET,
  created_at TIMESTAMP WITH TIME ZONE DEFAULT NOW()
);
```

#### `alert_history` (알람 발송 이력)
```sql
CREATE TABLE alert_history (
  id UUID PRIMARY KEY DEFAULT uuid_generate_v4(),
  alert_type TEXT NOT NULL, -- 'cpu', 'ram', 'disk', 'error', 'traffic'
  message TEXT NOT NULL,
  metrics JSONB, -- 발생 당시 메트릭 값
  sent_at TIMESTAMP WITH TIME ZONE DEFAULT NOW(),
  resolved_at TIMESTAMP WITH TIME ZONE
);
```

### 5.3 시스템 상태 및 모니터링

#### 5.3.1 서버 리소스 모니터링

**백엔드 API (Node.js)**
```javascript
// server/monitoring/server.js
const express = require('express');
const os = require('os');
const diskusage = require('diskusage');
const { createClient } = require('@supabase/supabase-js');

const app = express();
const supabase = createClient(process.env.SUPABASE_URL, process.env.SUPABASE_KEY);

// 5분마다 메트릭 수집 (초기 트래픽 고려하여 여유있게 설정)
setInterval(async () => {
  const cpuUsage = getCPUUsage();
  const ramUsage = (1 - (os.freemem() / os.totalmem())) * 100;
  const disk = await diskusage.check('/');
  const diskUsage = (disk.used / disk.total) * 100;

  const metrics = {
    cpu_usage: cpuUsage,
    ram_usage: ramUsage,
    ram_total: os.totalmem(),
    ram_used: os.totalmem() - os.freemem(),
    disk_usage: diskUsage,
    disk_total: disk.total,
    disk_used: disk.used,
    network_rx: getNetworkStats().rx,
    network_tx: getNetworkStats().tx
  };

  await supabase.from('server_metrics').insert(metrics);

  // 임계치 체크 및 알람 발송
  checkThresholdsAndAlert(metrics);
}, 300000);

app.listen(8002, () => console.log('Monitoring server running on port 8002'));
```

**임계치 체크 및 알람:**
```javascript
const checkThresholdsAndAlert = async (metrics) => {
  const { data: settings } = await supabase
    .from('admin_settings')
    .select('*')
    .single();

  const alerts = [];

  if (metrics.cpu_usage > settings.cpu_threshold) {
    alerts.push({
      type: 'cpu',
      message: `CPU 사용량이 ${metrics.cpu_usage.toFixed(2)}%로 임계치(${settings.cpu_threshold}%)를 초과했습니다.`
    });
  }

  if (metrics.ram_usage > settings.ram_threshold) {
    alerts.push({
      type: 'ram',
      message: `메모리 사용량이 ${metrics.ram_usage.toFixed(2)}%로 임계치(${settings.ram_threshold}%)를 초과했습니다.`
    });
  }

  if (metrics.disk_usage > settings.disk_threshold) {
    alerts.push({
      type: 'disk',
      message: `디스크 사용량이 ${metrics.disk_usage.toFixed(2)}%로 임계치(${settings.disk_threshold}%)를 초과했습니다.`
    });
  }

  // 중복 알람 방지: 최근 1시간 이내에 같은 타입 알람이 있는지 확인
  for (const alert of alerts) {
    const { data: recentAlert } = await supabase
      .from('alert_history')
      .select('*')
      .eq('alert_type', alert.type)
      .is('resolved_at', null)
      .gte('sent_at', new Date(Date.now() - 3600000).toISOString())
      .single();

    if (!recentAlert) {
      await sendEmailAlert(alert, settings.alert_email);
      await supabase.from('alert_history').insert({
        alert_type: alert.type,
        message: alert.message,
        metrics: metrics
      });
    }
  }
};
```

#### 5.3.2 트래픽 대역폭

**네트워크 모니터링:**
```javascript
// 10분 간격으로 네트워크 I/O 측정 (초기 트래픽 고려)
const networkStats = {
  lastRx: 0,
  lastTx: 0
};

const getNetworkStats = () => {
  const interfaces = os.networkInterfaces();
  let totalRx = 0, totalTx = 0;

  // eth0 또는 주요 네트워크 인터페이스 통계 수집
  // (실제로는 /proc/net/dev 파일 파싱 또는 라이브러리 사용)

  const rxPerSecond = totalRx - networkStats.lastRx;
  const txPerSecond = totalTx - networkStats.lastTx;

  networkStats.lastRx = totalRx;
  networkStats.lastTx = totalTx;

  return { rx: rxPerSecond, tx: txPerSecond };
};
```

**임계치 알람:**
- 10분당 전송량이 설정된 임계치 초과 시 알람
- 예: 5GB/10분 초과 시 이메일 발송

```nginx
# 별도 스크립트로 Nginx 에러 로그 파싱
```

**에러 로그 파싱 스크립트:**
```javascript
// server/monitoring/errorLogParser.js
const fs = require('fs');
const readline = require('readline');
const { createClient } = require('@supabase/supabase-js');

const supabase = createClient(process.env.SUPABASE_URL, process.env.SUPABASE_KEY);

const parseNginxErrorLog = (line) => {
  // Nginx 에러 로그 형식 파싱
  const match = line.match(/\[error\] .* client: ([\d.]+), .* request: "([^"]+)"/);

  if (match) {
    return {
      ip_address: match[1],
      page_url: match[2],
      message: line
    };
  }
  return null;
};

// tail -f 방식으로 실시간 모니터링
const tailErrorLog = () => {
  const stream = fs.createReadStream('/var/log/nginx/pulspia-error.log', {
    encoding: 'utf8',
    start: fs.statSync('/var/log/nginx/pulspia-error.log').size // 파일 끝부터 시작
  });

  stream.on('data', async (chunk) => {
    const lines = chunk.split('\n');
    for (const line of lines) {
      const parsed = parseNginxErrorLog(line);
      if (parsed) {
        await supabase.from('system_logs').insert({
          log_type: 'error',
          ...parsed
        });

        // 에러 발생 시 이메일 알람
        await sendErrorAlert(parsed);
      }
    }
  });
};
```

**에러 알람:**
- 500 에러 발생 시 즉시 이메일 발송

#### 5.3.4 데이터베이스 상태

**Supabase DB 연결 체크:**
```javascript
const checkDatabaseHealth = async () => {
  try {
    const start = Date.now();
    const { data, error } = await supabase
      .from('admin_settings')
      .select('id')
      .limit(1)
      .single();

    const queryTime = Date.now() - start;

    if (error) {
      throw new Error('Database connection failed');
    }

    // Slow Query 감지 (1초 이상)
    if (queryTime > 1000) {
      await sendEmailAlert({
        type: 'database',
        message: `데이터베이스 응답 시간이 ${queryTime}ms로 느립니다.`
      });
    }

    return { connected: true, queryTime };
  } catch (error) {
    await sendEmailAlert({
      type: 'database',
      message: `데이터베이스 연결 실패: ${error.message}`
    });
    return { connected: false, error: error.message };
  }
};

// 15분마다 헬스 체크 (초기 단계 트래픽 고려)
setInterval(checkDatabaseHealth, 900000);
```

1. 모니터링 서버에서 CPU 사용량 85% 감지
2. 임계치(80%) 초과 확인
3. 알람 이력 테이블에서 최근 1시간 내 중복 알람 확인
4. 중복 없으면 EmailJS로 이메일 발송
5. 알람 이력 테이블에 기록
6. 관리자 페이지에 알람 뱃지 표시

#### 시나리오 3: 에러 로그 수집 및 표시
1. 프론트엔드에서 500 에러 발생
2. ErrorBoundary에서 캡처
3. Supabase `system_logs` 테이블에 저장
4. 관리자 페이지 에러 로그 테이블에 실시간 표시
5. 500 에러이므로 즉시 이메일 알람 발송

   - 차단 해제
   - 목록에서 제거 확인

   - 사이트 이름 변경
